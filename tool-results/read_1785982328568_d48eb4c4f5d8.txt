    53→  const [loading, setLoading] = React.useState(false)

    65→      setMessageIndex((i) => (i + 1) % LOADING_MESSAGES.length)

    67→    return () => clearInterval(id)
    68→  }, [loading])
    69→
    70→  async function handleSubmit(e: React.FormEvent) {
    71→    e.preventDefault()
    72→    const trimmed = account.trim()
    73→    if (!trimmed) {
    74→      toast({
    75→        title: "حقل مطلوب",
    76→        description: "الرجاء إدخال حساب المنافس للبدء.",
    77→        variant: "destructive",
    78→      })
    79→      return

    81→    if (!platform) {
    82→      toast({
    83→        title: "اختر منصة",
    84→        description: "الرجاء اختيار المنصة الاجتماعية المناسبة.",
    85→        variant: "destructive",

    96→        method: "POST",

   112→        title: "فشل التحليل",

   121→  const topKeyword = result?.keywords?.[0]?.keyword
   122→
   123→  return (
   124→    <div className="space-y-8">
   125→      {/* Section Header */}

   127→        <h2 className="font-display text-xl font-bold flex items-center gap-2">
   128→          <Users className="w-5 h-5 text-primary" />
   129→          تحليل المنافسين

   131→        <p className="text-sm text-muted-foreground">

   135→      </header>

   139→        <div className="absolute top-0 right-0 left-0 h-1 bg-gradient-brand" />
   140→        <CardContent className="pt-6">
   141→          <form onSubmit={handleSubmit} className="space-y-5">
   142→            {/* Account input with User icon prefix */}
   143→            <div className="space-y-2">
   144→              <label
   145→                htmlFor="competitor-account"
   146→                className="text-sm font-semibold flex items-center gap-1.5"
   147→              >
   148→                <User className="w-3.5 h-3.5 text-primary" />
   149→                حساب المنافس
   150→              </label>
   151→              <div className="relative">
   152→                <span className="absolute inset-y-0 right-3 flex items-center pointer-events-none text-muted-foreground">
   153→                  <AtSign />
   154→                </span>
   155→                <Input
   156→                  id="competitor-account"
   157→                  value={account}
   158→                  onChange={(e) => setAccount(e.target.value)}
   159→                  placeholder="username"
   160→                  autoComplete="off"
   161→                  disabled={loading}

   164→                />
   165→              </div>
   166→              <p
   167→                id="competitor-account-help"
   168→                className="text-[11px] text-muted-foreground"
   169→              >
   170→                أدخل اسم المستخدم بدون @ — مثال: competitors_handle
   171→              </p>
   172→            </div>

   178→                اختر المنصة

   180→              <div className="grid grid-cols-2 sm:grid-cols-4 gap-2.5">
   181→                {PLATFORM_LIST.map((p) => {

   192→                        selected

   194→                          : "bg-card text-foreground border-border hover:border-primary/40 hover:bg-accent/50"
   195→                      )}
   196→                    >
   197→                      <PlatformBadge
   198→                        platform={p.id}
   199→                        size="sm"
   200→                        className={cn(selected && "[&_div]:ring-2 [&_div]:ring-white/60")}
   201→                      />

   203→                    </button>
   204→                  )
   205→                })}
   206→              </div>
   207→            </div>
   208→
   209→            {/* Submit */}
   210→            <div className="flex flex-col-reverse sm:flex-row sm:items-center sm:justify-between gap-3 pt-1">
   211→              <p className="text-[11px] text-muted-foreground flex items-center gap-1.5">

   222→                  <>
   223→                    <Loader2 className="w-4 h-4 animate-spin" />
   224→                    جاري التحليل...
   225→                  </>
   226→                ) : (
   227→                  <>
   228→                    <Search className="w-4 h-4" />
   229→                    تحليل الحساب
   230→                  </>
   231→                )}
   232→              </Button>
   233→            </div>
   234→          </form>
   235→        </CardContent>
   236→      </Card>
   237→
   238→      {/* Loading State */}
   239→      {loading && <LoadingState messageIndex={messageIndex} />}
   240→
   241→      {/* Error State */}
   242→      {!loading && error && (
   243→        <Card className="border-rose-200 dark:border-rose-900/60 bg-rose-50/50 dark:bg-rose-950/20">
   244→          <CardContent className="pt-6">
   245→            <div className="flex items-start gap-3">
   246→              <div className="shrink-0 w-10 h-10 rounded-full bg-rose-100 dark:bg-rose-900/40 flex items-center justify-center">
   247→                <AlertCircle className="w-5 h-5 text-rose-600 dark:text-rose-400" />
   248→              </div>
   249→              <div className="flex-1 min-w-0">
   250→                <h3 className="font-bold text-rose-700 dark:text-rose-300 mb-1">
   251→                  تعذّر إكمال التحليل
   252→                </h3>
   253→                <p className="text-sm text-rose-600/90 dark:text-rose-400/90 leading-relaxed">
   254→                  {error}
   255→                </p>
   256→                <Button
   257→                  variant="outline"
   258→                  size="sm"
   259→                  className="mt-3 rounded-lg border-rose-200 dark:border-rose-900/60 text-rose-700 dark:text-rose-300 hover:bg-rose-50 dark:hover:bg-rose-950/40"
   260→                  onClick={() => setError(null)}
   261→                >
   262→                  المحاولة مرة أخرى
   263→                </Button>
   264→              </div>
   265→            </div>
   266→          </CardContent>
   267→        </Card>
   268→      )}
   269→
   270→      {/* Empty State (before first search) */}
   271→      {!loading && !error && !result && <EmptyState />}
   272→
   273→      {/* Results */}
   274→      {!loading && !error && result && (
   275→        <Results data={result} onNavigate={onNavigate} topKeyword={topKeyword} />
   276→      )}
   277→    </div>
   278→  )
   279→}
