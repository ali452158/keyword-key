   112→   112→/**
   113→   113→ * Group trends by keyword, summing each platform's volume. Returns the
   114→   114→ * top 5 keywords by total volume. Missing platform => 0.
   115→   115→ */
   116→   116→function buildBarChartData(trends: KeywordTrend[]): BarDatum[] {
   117→   117→  const grouped = new Map<string, BarDatum>()
   118→   118→  for (const t of trends) {
   119→   119→    let entry = grouped.get(t.keyword)
   120→   120→    if (!entry) {
   121→   121→      entry = {
   122→   122→        keyword: t.keyword,
   123→   123→        tiktok: 0,
   124→   124→        youtube: 0,
   125→   125→        instagram: 0,
   126→   126→        facebook: 0,
   127→   127→      }
   128→   128→      grouped.set(t.keyword, entry)
   129→   129→    }
   130→   130→    entry[t.platform] += t.searchVolume
   131→   131→  }
   132→   132→  return Array.from(grouped.values())
   133→   133→    .map((d) => ({
   134→   134→      datum: d,
   135→   135→      total: d.tiktok + d.youtube + d.instagram + d.facebook,
   136→   136→    }))
   137→   137→    .sort((a, b) => b.total - a.total)
   138→   138→    .slice(0, 5)
   139→   139→    .map((x) => x.datum)
   140→   140→}
   141→   141→
   142→   142→/**
   143→   143→ * Build 7 days of stacked area data. For each platform we take its total
   144→   144→ * volume from the trends, divide by 7 to get a daily baseline, then apply
   145→   145→ * a per-day variation factor.
   146→   146→ */
   147→   147→function buildAreaChartData(trends: KeywordTrend[]): AreaDatum[] {
   148→   148→  const totals: Record<Platform, number> = {
   149→   149→    tiktok: 0,
   150→   150→    youtube: 0,
   151→   151→    instagram: 0,
   152→   152→    facebook: 0,
   153→   153→  }
   154→   154→  for (const t of trends) totals[t.platform] += t.searchVolume
   155→   155→
   156→   156→  return ARABIC_DAYS.map((day, i) => {
   157→   157→    const f = DAILY_FACTORS[i]
   158→   158→    return {
   159→   159→      day,
   160→   160→      tiktok: Math.round((totals.tiktok / 7) * f),
   161→   161→      youtube: Math.round((totals.youtube / 7) * f),
   162→   162→      instagram: Math.round((totals.instagram / 7) * f),
   163→   163→      facebook: Math.round((totals.facebook / 7) * f),
   164→   164→    }
   165→   165→  })
   166→   166→}
   167→   167→
   168→   168→/**
   169→   169→ * Count how many trends belong to each platform (count > 0 only).
   170→   170→ */
   171→   171→function buildPieChartData(trends: KeywordTrend[]): PieDatum[] {
   172→   172→  const counts: Record<Platform, number> = {
   173→   173→    tiktok: 0,
   174→   174→    youtube: 0,
   175→   175→    instagram: 0,
   176→   176→    facebook: 0,
   177→   177→  }
   178→   178→  for (const t of trends) counts[t.platform]++
   179→   179→  return PLATFORMS_FOR_CHART.filter((p) => counts[p] > 0).map((p) => ({
   180→   180→    name: PLATFORMS[p].arabicName,
   181→   181→    value: counts[p],
   182→   182→    color: PLATFORM_COLORS[p],
   183→   183→    platform: p,
   184→   184→  }))
   185→   185→}
   186→   186→
   187→   187→function ChartTooltip({ active, payload, label }: any) {
   188→   188→  if (!active || !payload?.length) return null
   189→   189→  return (
   190→   190→    <div className="bg-popover border border-border rounded-lg p-2 shadow-md text-xs">
   191→   191→      {label && <p className="font-semibold mb-1">{label}</p>}
   192→   192→      {payload.map((p: any, i: number) => (
   193→   193→        <p key={i} className="flex items-center gap-1.5">
   194→   194→          <span
   195→   195→            className="w-2 h-2 rounded-full"
   196→   196→            style={{ background: p.color || p.fill }}
   197→   197→          />
   198→   198→          <span className="text-muted-foreground">{p.name}:</span>
   199→   199→          <span className="font-semibold">{formatNumber(p.value)}</span>
   200→   200→        </p>
   201→   201→      ))}
   202→   202→    </div>
   203→   203→  )
   204→   204→}
