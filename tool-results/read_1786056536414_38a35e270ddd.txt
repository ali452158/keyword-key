    62→  const [loading, setLoading] = React.useState(false)

    65→
    66→  const [copiedId, setCopiedId] = React.useState<string | null>(null)

    68→
    69→  // Related trends state
    70→  const [allTrends, setAllTrends] = React.useState<KeywordTrend[]>([])
    71→  const [trendsLoading, setTrendsLoading] = React.useState(false)
    72→  const [trendsError, setTrendsError] = React.useState(false)
    73→
    74→  // Fetch trends from /api/trending?period=daily&limit=20
    75→  const fetchTrends = React.useCallback(async function fetchTrends() {
    76→    setTrendsLoading(true)
    77→    setTrendsError(false)
    78→    try {
    79→      const res = await fetch(
    80→        "/api/trending?period=daily&limit=20",
    81→        { cache: "no-store" }
    82→      )
    83→      const json = await res.json()
    84→      if (!res.ok || !json.success) {
    85→        throw new Error(json?.error || "فشل تحميل الترندات")
    86→      }
    87→      const data: KeywordTrend[] = json.data ?? []
    88→      setAllTrends(data)
    89→    } catch {
    90→      setTrendsError(true)
    91→    } finally {
    92→      setTrendsLoading(false)
    93→    }
    94→  }, [])
    95→
    96→  // Fetch trends on mount (once) so we always have data ready
    97→  const didFetchOnMount = React.useRef(false)
    98→  React.useEffect(() => {
    99→    if (didFetchOnMount.current) return
   100→    didFetchOnMount.current = true
   101→    void fetchTrends()
   102→  }, [fetchTrends])
   103→
   104→  // Debounced keyword change -> refetch trends (500ms)
   105→  const debounceRef = React.useRef<ReturnType<typeof setTimeout> | null>(null)
   106→  React.useEffect(() => {
   107→    if (debounceRef.current) {
   108→      clearTimeout(debounceRef.current)
   109→    }
   110→    debounceRef.current = setTimeout(() => {
   111→      void fetchTrends()
   112→    }, 500)
   113→    return () => {
   114→      if (debounceRef.current) {
   115→        clearTimeout(debounceRef.current)
   116→      }
   117→    }
   118→    // Re-fetch when keyword changes (debounced). fetchTrends is stable.
   119→  }, [keyword, fetchTrends])
   120→
   121→  // Compute related trends based on the current keyword
   122→  const relatedTrends = React.useMemo<KeywordTrend[]>(() => {
   123→    const kw = keyword.trim().toLowerCase()
   124→    if (!allTrends.length) return []
   125→    if (!kw) {
   126→      // No keyword: show top trends sorted by trendScore
   127→      return [...allTrends]
   128→        .sort((a, b) => b.trendScore - a.trendScore)
   129→        .slice(0, 8)
   130→    }
   131→    // Match by keyword text contains OR category contains OR hashtag
   132→    const matches = allTrends.filter((t) => {
   133→      const tKw = t.keyword.toLowerCase()
   134→      const tCat = (t.category || "").toLowerCase()
   135→      const tHash = (t.hashtag || "").toLowerCase()
   136→      return (
   137→        tKw.includes(kw) ||
   138→        kw.includes(tKw) ||
   139→        tCat.includes(kw) ||
   140→        kw.includes(tCat) ||
   141→        tHash.includes(kw)
   142→      )
   143→    })
   144→    if (matches.length > 0) {
   145→      return matches
   146→        .sort((a, b) => b.trendScore - a.trendScore)
   147→        .slice(0, 8)
   148→    }
   149→    // No direct match: fall back to top trends
   150→    return [...allTrends]
   151→      .sort((a, b) => b.trendScore - a.trendScore)
   152→      .slice(0, 8)
   153→  }, [allTrends, keyword])
   154→
   155→  // Match an idea to a trending keyword
   156→  const findIdeaTrend = React.useCallback(
   157→    function findIdeaTrend(idea: ContentIdea): KeywordTrend | undefined {
   158→      if (!relatedTrends.length) return undefined
   159→      const title = (idea.title || "").toLowerCase()
   160→      const hashText = (idea.hashtags || []).join(" ").toLowerCase()
   161→      const haystack = `${title} ${hashText}`
   162→      // Find a trend whose keyword appears in the idea text
   163→      const matched = relatedTrends.find((t) => {
   164→        const tKw = t.keyword.toLowerCase()
   165→        const tHash = (t.hashtag || "").toLowerCase()
   166→        return (
   167→          (tKw.length > 2 && haystack.includes(tKw)) ||
   168→          (tHash.length > 2 && haystack.includes(tHash))
   169→        )
   170→      })
   171→      if (matched) return matched
   172→      // Fall back: deterministic-ish pick by hashing idea title
   173→      const idx =
   174→        Math.abs(
   175→          [...idea.title].reduce(
   176→            (acc, ch) => acc + ch.charCodeAt(0),
   177→            0
   178→          )
   179→        ) % relatedTrends.length
   180→      return relatedTrends[idx]
   181→    },
   182→    [relatedTrends]
   183→  )
   184→
   185→  // Auto-generate when navigated with initialKeyword
   186→  const didAutoGen = React.useRef(false)
   187→  React.useEffect(() => {
   188→    if (initialKeyword && initialKeyword.trim() && !didAutoGen.current) {
   189→      didAutoGen.current = true
   190→      void generate(initialKeyword.trim(), platform, count)
   191→    }
   192→    // Intentionally run once on mount when initialKeyword is provided
   193→  }, [initialKeyword])
   194→
   195→  // Rotate loading messages while waiting
   196→  React.useEffect(() => {
   197→    if (!loading) return
   198→    setLoadingMsgIdx(0)
   199→    const id = window.setInterval(() => {
   200→      setLoadingMsgIdx((i) => (i + 1) % LOADING_MESSAGES.length)
   201→    }, 2800)
   202→    return () => window.clearInterval(id)
   203→  }, [loading])
   204→
   205→  async function generate(kw: string, plat: Platform, cnt: number) {
   206→    setLoading(true)
   207→    try {
   208→      const res = await fetch("/api/content/generate", {
   209→        method: "POST",
   210→        headers: { "Content-Type": "application/json" },
   211→        body: JSON.stringify({ keyword: kw, platform: plat, count: cnt }),
   212→      })
   213→      const json = await res.json()
   214→      if (!res.ok || !json.success) {
   215→        throw new Error(json?.error || "فشل توليد الأفكار")
   216→      }
   217→      const data: ContentIdea[] = json.data ?? []
   218→      setIdeas(data)
   219→      setHasGenerated(true)
   220→      toast({
   221→        title: "تم توليد الأفكار بنجاح",
   222→        description: `${data.length} فكرة محتوى جاهزة للاستخدام`,
   223→      })
   224→    } catch (err) {
   225→      const message =
   226→        err instanceof Error ? err.message : "حدث خطأ غير متوقع"
   227→      toast({
   228→        title: "تعذّر توليد الأفكار",
   229→        description: message,
   230→        variant: "destructive",
   231→      })
   232→      setIdeas([])
   233→      setHasGenerated(true)
   234→    } finally {
   235→      setLoading(false)
   236→    }
   237→  }

   239→  function handleSubmit(e: React.FormEvent) {

   257→    }
   258→    void generate(trimmed, platform, count)

   363→                      disabled={loading}

   369→                      )}
   370→                    >
   371→                      <PlatformBadge
   372→                        platform={p.id}
   373→                        size="sm"

   375→                      />
   376→                      <span>{p.arabicName}</span>
   377→                    </button>
   378→                  )
   379→                })}
   380→              </div>
   381→            </div>
   382→
   383→            {/* Count selector */}
   384→            <div className="space-y-2">
   385→              <span className="text-sm font-semibold flex items-center gap-1.5">
   386→                <Film className="w-4 h-4 text-primary" />
   387→                عدد الأفكار

   398→                      key={n}
   399→                      type="button"

   401→                      aria-checked={active}
   402→                      onClick={() => setCount(n)}

   405→                        "px-5 py-1.5 rounded-md text-sm font-semibold transition-all min-w-[64px]",
   406→                        active
   407→                          ? "bg-gradient-brand text-white shadow-sm"
   408→                          : "text-muted-foreground hover:text-foreground hover:bg-accent"
   409→                      )}
   410→                    >
   411→                      {n} أفكار
   412→                    </button>
   413→                  )
   414→                })}
   415→              </div>
   416→            </div>
   417→
   418→            {/* Submit */}
   419→            <div className="flex flex-col sm:flex-row sm:items-center gap-3 pt-1">

   432→                    <Sparkles className="w-4 h-4" />

   435→                )}
   436→              </Button>
   437→              <p className="text-xs text-muted-foreground flex items-center gap-1.5">

   451→      {!loading && !hasGenerated && ideas.length === 0 && (

   458→
   459→      {showResults && (

   461→          {/* Ideas — 2 cols on lg */}
   462→          <div className="lg:col-span-2 space-y-4">

   465→                <h3 className="font-display text-lg font-bold flex items-center gap-2">
   466→                  <Lightbulb className="w-5 h-5 text-primary" />
   467→                  أفكار مقترحة
   468→                  <Badge
   469→                    variant="secondary"
   470→                    className="bg-primary/10 text-primary border border-primary/20"
   471→                  >
   472→                    {ideas.length}
   473→                  </Badge>
   474→                </h3>
   475→                <Button
   476→                  variant="ghost"
   477→                  size="sm"
   478→                  className="text-muted-foreground hover:text-primary"
   479→                  onClick={() => {
   480→                    setIdeas([])
   481→                    setHasGenerated(false)
   482→                  }}
   483→                >
   484→                  مسح
   485→                </Button>
   486→              </div>
   487→
   488→              <div className="grid grid-cols-1 md:grid-cols-2 gap-4">

   492→                    <IdeaCard
   493→                      key={`${idea.title}-${idx}`}

   495→                      index={idx}
   496→                      copied={copiedId === `${idea.title}-${idx}`}
   497→                      onCopy={() => handleCopy(idea, `${idea.title}-${idx}`)}
   498→                      relatedTrend={trend}
   499→                      onNavigate={onNavigate}
   500→                    />
   501→                  )
   502→                })}
   503→              </div>
   504→            </section>
   505→          </div>
   506→
   507→          {/* Related Trends sidebar — 1 col on lg */}
   508→          <aside className="lg:sticky lg:top-4 lg:self-start">
   509→            <RelatedTrendsCard
   510→              trends={relatedTrends}
   511→              loading={trendsLoading}
   512→              error={trendsError}
   513→              onRefresh={() => void fetchTrends()}
   514→              onTrendClick={handleTrendClick}
   515→              currentKeyword={keyword.trim()}
   516→            />
   517→          </aside>
   518→        </div>
   519→      )}
   520→    </div>
   521→  )
   522→}
